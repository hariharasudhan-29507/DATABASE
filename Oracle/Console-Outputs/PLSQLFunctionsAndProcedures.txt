24.) PROCEDURE TO DISPLAY THE MEMBER WITH HIGHEST AGE IN EACH CITY
     (NESTED WHILE LOOP)

SQL> CREATE OR REPLACE PROCEDURE max_age_per_city
  2  IS
  3      CURSOR city_cur IS
  4          SELECT DISTINCT ADDRESS_CITY FROM MEMBER WHERE ADDRESS_CITY IS NOT NULL;
  5
  6      v_city     MEMBER.ADDRESS_CITY%TYPE;

  8      v_mem_name MEMBER.MEMBER_NAME%TYPE;
  9  BEGIN
 10      OPEN city_cur;

 12

 19          FROM MEMBER

 24                  SELECT MEMBER_NAME FROM MEMBER
 25                  WHERE ADDRESS_CITY = v_city AND AGE = v_max_age;

 35              CLOSE mem_cur;
 36          END;
