  2  SELECT p.PAYMENT_ID, p.STATUS, SUM(p.AMOUNT) AS TOTAL_AMOUNT

6)to create view high value branch from another view high payment

SQL> CREATE VIEW branch_payment_total AS
  2  SELECT b.BRANCH_NAME, SUM(p.AMOUNT) AS BRANCH_TOTAL
  3  FROM PAYMENT p
  4  JOIN BRANCH b ON p.BRANCH_ID = b.BRANCH_ID
  5  GROUP BY b.BRANCH_NAME;
